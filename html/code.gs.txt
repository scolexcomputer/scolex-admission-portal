const SHEET_NAME = "Applications";
const PHOTO_FOLDER_ID = "1BNqDro8u5yiCujiF1O4nDTP-ze8XUIBz";
const SIGNATURE_FOLDER_ID = "1vNVcHtpaxN58Tm0KZ-ioI_aKZzTioezT";

// =============================
// Handle POST Request
// =============================

function doPost(e) {

try {


const sheet = SpreadsheetApp
.getActiveSpreadsheet()
.getSheetByName(SHEET_NAME);


// Generate Application ID

const applicationID = "APP" + Utilities.formatDate(
new Date(),
Session.getScriptTimeZone(),
"yyyyMMddHHmmss"
);


// Upload Photo

let photoURL = "";


if(e.parameter.photo){


const folder =
DriveApp.getFolderById(PHOTO_FOLDER_ID);



const blob = Utilities.newBlob(

Utilities.base64Decode(e.parameter.photo),

e.parameter.mimeType,

e.parameter.fileName

);



const file = folder.createFile(blob);


photoURL = file.getUrl();


}
//upload signature
let signURL = "";

if (e.parameter.signature) {

  const folder = DriveApp.getFolderById(SIGNATURE_FOLDER_ID);

  const blob = Utilities.newBlob(
    Utilities.base64Decode(e.parameter.signature),
    e.parameter.signatureType,
    e.parameter.signatureName
  );

  const file = folder.createFile(blob);

  signURL = file.getUrl();

}

// Current Timestamp

const timestamp = new Date();



// Save Data

sheet.appendRow([

applicationID,

timestamp,

e.parameter.fullname,

e.parameter.mobile,

e.parameter.email,

e.parameter.dob,

e.parameter.gender,

e.parameter.education,

e.parameter.state,

e.parameter.courses,

e.parameter.contact,

e.parameter.address,

photoURL,

signURL

]);
const lastRow =
sheet.getLastRow();


const rowData =
sheet.getRange(lastRow,1,1,14)
.getValues()[0];


const pdfURL =
createPDF(rowData);


return ContentService
.createTextOutput("SUCCESS")
.setMimeType(ContentService.MimeType.TEXT);



}

catch(err){


return ContentService
.createTextOutput(
"ERROR: " + err.toString()
)
.setMimeType(ContentService.MimeType.TEXT);


}

}
function createPDF(rowData){

const folder =
DriveApp.getFolderById("1gLDZ7MC0AwPxe4Jb_OXhPCAAgFU5LQJu");


// Open Template Sheet

const ss =
SpreadsheetApp.getActiveSpreadsheet();

const template =
ss.getSheetByName("PDF_Template");


// Replace data
template.getRange("B4").setValue(rowData[0]);   // Application ID
template.getRange("B5").setValue(rowData[2]);   // Full Name
template.getRange("B6").setValue(rowData[3]);   // Mobile
template.getRange("B7").setValue(rowData[4]);   // Email
template.getRange("B8").setValue(rowData[5]);   // DOB
template.getRange("B9").setValue(rowData[6]);   // Gender
template.getRange("B10").setValue(rowData[7]);  // Education
template.getRange("B11").setValue(rowData[8]);  // State
template.getRange("B12").setValue(rowData[9]);  // Course
template.getRange("B13").setValue(rowData[10]); // Contact
template.getRange("B14").setValue(rowData[11]); // Address

// Student Photo
if (rowData[12] != "") {

  var fileId = rowData[12].match(/[-\w]{25,}/)[0];

  var imageUrl =
    "https://drive.google.com/uc?export=view&id=" + fileId;

  template.getRange("D4").clearContent();

  template.getRange("D4").setFormula(
    '=IMAGE("' + imageUrl + '",4,180,140)'
  );

}
// Student Signature
if (rowData[13] != "") {

  var signId = rowData[13].match(/[-\w]{25,}/)[0];

  var signImageUrl =
      "https://drive.google.com/uc?export=view&id=" + signId;

  template.getRange("D12").clearContent();   // Change cell as needed

  template.getRange("D12").setFormula(
      '=IMAGE("' + signImageUrl + '",4,35,200)'
  );

}

// Wait for sheet update
SpreadsheetApp.flush();
Utilities.sleep(3000);

// Export PDF
const url =
"https://docs.google.com/spreadsheets/d/" +
ss.getId() +
"/export?format=pdf&gid=" +
template.getSheetId();

const token =
ScriptApp.getOAuthToken();

const response =
UrlFetchApp.fetch(url, {
  headers: {
    Authorization: "Bearer " + token
  }
});

const pdf =
folder.createFile(
  response.getBlob()
    .setName(
      rowData[0] + "_Admission.pdf"
    )
);

return pdf.getUrl();
}
